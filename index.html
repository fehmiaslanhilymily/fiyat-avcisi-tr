1) `scraper.py` (2 Saatte Bir Çalışan Python Tarayıcı)

Ortalama geçmiş fiyatın %60 altına inen ürünleri otomatik "Hatalı Fiyat" olarak işaretler.

import json
import os
import time
import urllib.request
from datetime import datetime, timezone

DATA_DIR = "data"
DEALS_FILE = os.path.join(DATA_DIR, "deals.json")
HISTORY_FILE = os.path.join(DATA_DIR, "price_history.json")

PLATFORMS = [
    "Trendyol", "Hepsiburada", "Amazon TR", "N11", "Pazarama", "Teknosa"
]

def load_json(path, default):
    if os.path.exists(path):
        with open(path, "r", encoding="utf-8") as f:
            try:
                return json.load(f)
            except Exception:
                return default
    return default

def save_json(path, payload):
    os.makedirs(os.path.dirname(path), exist_ok=True)
    with open(path, "w", encoding="utf-8") as f:
        json.dump(payload, f, ensure_ascii=False, indent=2)

def fetch_platform_candidates():
    # Buraya platform API / feed veya çerezli istek fonksiyonlarınızı bağlayabilirsiniz.
    # Örnek yapı: her ürün {id, platform, title, current_price, old_price, url} döndürür.
    return []

def classify_deal(item, history):
    current = float(item["current_price"])
    old = float(item.get("old_price") or current)
    prev_prices = history.get(item["id"], [])
    avg_price = (sum(prev_prices) / len(prev_prices)) if prev_prices else old

    discount_pct = round(((avg_price - current) / avg_price) * 100, 1) if avg_price > 0 else 0
    is_error_price = discount_pct >= 60.0

    prev_prices.append(current)
    history[item["id"]] = prev_prices[-15:]

    return {
        "id": item["id"],
        "platform": item["platform"],
        "title": item["title"],
        "current_price": current,
        "old_price": old,
        "avg_price": round(avg_price, 2),
        "discount_pct": max(discount_pct, 0),
        "deal_type": "ERROR_PRICE" if is_error_price else "NORMAL_DEAL",
        "url": item.get("url", "#")
    }

def main():
    history = load_json(HISTORY_FILE, {})
    raw_items = fetch_platform_candidates()
    deals = []

    for item in raw_items:
        if float(item["current_price"]) < float(item.get("old_price", item["current_price"])):
            deals.append(classify_deal(item, history))
        time.sleep(0.2)

    deals.sort(key=lambda x: x["discount_pct"], reverse=True)

    output = {
        "last_updated": datetime.now(timezone.utc).strftime("%Y-%m-%d %H:%M UTC"),
        "total": len(deals),
        "deals": deals
    }

    save_json(HISTORY_FILE, history)
    save_json(DEALS_FILE, output)
    print(f"Güncellendi: {len(deals)} fırsat kaydedildi.")

if __name__ == "__main__":
    main()
2) `.github/workflows/scrape.yml` (Her 2 Saatte Bir Otomatik Çalıştırma)

GitHub Actions üzerinde 00:00, 02:00, 04:00... saatlerinde çalışır ve `data/deals.json` dosyasını günceller.

name: FiyatAvcısı TR - 2 Saatlik Otomatik Tarama

on:
  schedule:
    - cron: '0 */2 * * *'
  workflow_dispatch:

permissions:
  contents: write

jobs:
  scrape-deals:
    runs-on: ubuntu-latest
    steps:
      - name: Depoyu İndir
        uses: actions/checkout@v4

      - name: Python Kurulumu
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'

      - name: Tarama Botunu Çalıştır
        run: python scraper.py

      - name: Değişiklikleri Depoya Kaydet
        run: |
          git config user.name "FiyatAvci Bot"
          git config user.email "bot@users.noreply.github.com"
          git add data/deals.json data/price_history.json
          git diff --staged --quiet || git commit -m "Otomatik Fiyat Güncellemesi: $(date -u +'%Y-%m-%d %H:%M UTC')"
          git push
